# D5-C CEP（复杂事件处理）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 支持跨事件时序规则（频率类、序列类含 NOT、聚合类），通过独立 Flink Job 检测模式命中，结果回写现有评估链路。

**Architecture:** `rule-cep-job`（独立 Flink 应用）消费 RocketMQ `rule-events` topic，检测三类 CEP 模式，命中后把 `CepFiredEvent` 发到 `rule-cep-fired` topic；`rule-eval-svc` 新增 `CepFiredEventConsumer` 消费该 topic，构造合成 `RuleEvent` 投入现有评估链路；`rule-config-svc` 追加 CEP 发布校验；`rule-api` 新增 `GET /api/v1/cep/rules` 端点供 Flink Job 启动时拉取规则。

**Tech Stack:** Java 25（Spring Boot 侧）、Apache Flink 1.20（CEP Job）、Apache Flink CEP 库（`flink-cep`）、RocketMQ Spring Boot Starter 2.3.x、Jackson

```java
// rule-config-svc/src/main/java/com/sstlfsj/rule/config/internal/publish/CepSpecValidator.java
package com.sstlfsj.rule.config.internal.publish;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.time.Duration;
import java.util.Set;

/** CEP 规则发布时的 conditionAst（CepRuleSpec JSON）字段校验。 */
class CepSpecValidator {

    private static final ObjectMapper MAPPER = new ObjectMapper();
    private static final Set<String> VALID_AGG_FUNCS =
            Set.of("SUM", "COUNT", "AVG", "MAX", "MIN");

    /**
     * 校验 CepRuleSpec JSON，不合法时抛 IllegalArgumentException。
     *
     * @param kind    规则 kind 字符串（CEP_FREQUENCY / CEP_SEQUENCE / CEP_AGGREGATION）
     * @param specJson conditionAst 字段内容（JSON 字符串）
     */
    static void validate(String kind, String specJson) {
        try {
            JsonNode node = MAPPER.readTree(specJson);
            switch (kind) {
                case "CEP_FREQUENCY" -> validateFrequency(node);
                case "CEP_SEQUENCE"  -> validateSequence(node);
                case "CEP_AGGREGATION" -> validateAggregation(node);
                default -> { /* 非 CEP kind，跳过 */ }
            }
        } catch (IllegalArgumentException e) {
            throw e;
        } catch (Exception e) {
            throw new IllegalArgumentException("CEP 规则 spec JSON 解析失败: " + e.getMessage());
        }
    }

    private static void validateFrequency(JsonNode n) {
        requireValidDuration(n, "windowSize");
        if (!n.hasNonNull("minCount") || n.get("minCount").asInt(0) <= 0) {
            throw new IllegalArgumentException("CEP_FREQUENCY: minCount 必须 > 0");
        }
    }

    private static void validateSequence(JsonNode n) {
        JsonNode steps = n.get("steps");
        if (steps == null || !steps.isArray() || steps.size() < 2) {
            throw new IllegalArgumentException("CEP_SEQUENCE: steps 至少需要 2 个");
        }
        for (JsonNode step : steps) {
            if (!step.hasNonNull("eventType") || step.get("eventType").asText().isBlank()) {
                throw new IllegalArgumentException("CEP_SEQUENCE: 每个 step 必须有 eventType");
            }
        }
        requireValidDuration(n, "withinWindow");
    }

    private static void validateAggregation(JsonNode n) {
        String func = n.hasNonNull("aggregateFunc") ? n.get("aggregateFunc").asText() : null;
        if (func == null || !VALID_AGG_FUNCS.contains(func)) {
            throw new IllegalArgumentException(
                    "CEP_AGGREGATION: aggregateFunc 必须是 " + VALID_AGG_FUNCS + " 之一，实际: " + func);
        }
        if (!n.hasNonNull("threshold")) {
            throw new IllegalArgumentException("CEP_AGGREGATION: threshold 不能为 null");
        }
    }

    private static void requireValidDuration(JsonNode n, String field) {
        if (!n.hasNonNull(field) || n.get(field).asText().isBlank()) {
            throw new IllegalArgumentException("CEP: " + field + " 必填");
        }
        try {
            Duration.parse(n.get(field).asText());
        } catch (Exception e) {
            throw new IllegalArgumentException(
                    "CEP: " + field + " 不是合法的 ISO 8601 duration: " + n.get(field).asText());
        }
    }
}
```

```java
        this.ruleVersionReadMapper = ruleVersionReadMapper;

    }
}
```

```java
import com.sstlfsj.rule.cep.model.CepRuleSpec;
import org.apache.flink.cep.pattern.Pattern;
import org.apache.flink.cep.pattern.conditions.SimpleCondition;

import java.time.Duration;
import java.util.Map;

/** 频率类 CEP Pattern 构建器：N 时间窗内同一主体触发 X 次即命中。 */
public class FrequencyPatternFactory {

    /**
     * 构建频率类 Pattern。
     * 使用 Flink CEP 的 times(minCount).within(windowSize) 语义。
     *
     * @param spec CEP 规则描述（kind=CEP_FREQUENCY）
     * @return Flink Pattern
     */
    public static Pattern<Map<String, Object>, Map<String, Object>> build(CepRuleSpec spec) {
        Duration window = Duration.parse(spec.windowSize);
        return Pattern.<Map<String, Object>>begin(spec.eventType + "_freq")
                .where(new SimpleCondition<>() {
                    @Override
                    public boolean filter(Map<String, Object> event) {
                        // eventType 匹配由上游 keyBy 后的 filter 保证，此处恒真
                        return true;
                    }
                })
                .times(spec.minCount)
                .within(org.apache.flink.streaming.api.windowing.time.Time.milliseconds(
                        window.toMillis()));
    }
}
```

```java
// rule-cep-job/src/main/java/com/sstlfsj/rule/cep/pattern/AggregationWindowFactory.java
package com.sstlfsj.rule.cep.pattern;

import com.sstlfsj.rule.cep.model.CepRuleSpec;
import com.sstlfsj.rule.cep.model.CepFiredEvent;
import org.apache.flink.api.common.functions.AggregateFunction;
import org.apache.flink.streaming.api.datastream.KeyedStream;
import org.apache.flink.streaming.api.datastream.SingleOutputStreamOperator;
import org.apache.flink.streaming.api.windowing.assigners.TumblingEventTimeWindows;
import org.apache.flink.streaming.api.windowing.time.Time;

import java.time.Duration;
import java.time.Instant;
import java.util.Map;

/**
 * 聚合类 Window 构建器：在 N 时间窗内对指定字段做 SUM/COUNT/AVG/MAX/MIN，
 * 超过 threshold 时输出 CepFiredEvent。
 */
public class AggregationWindowFactory {

    /**
     * 解析 spec.windowSize（ISO 8601 duration）为 Flink Time。
     */
    public static Time parseWindowSize(CepRuleSpec spec) {
        Duration d = Duration.parse(spec.windowSize);
        return Time.milliseconds(d.toMillis());
    }

    /**
     * 对已按 (tenantId:sceneCode:subjectId) keyBy 的流应用滚动窗口聚合。
     * 超过 threshold 时发出 CepFiredEvent。
     *
     * @param keyed 按主体 keyBy 后的事件流，元素为 RuleEvent payload Map
     * @param spec  CEP 聚合规则描述
     * @return 命中事件流
     */
    public static SingleOutputStreamOperator<CepFiredEvent> apply(
            KeyedStream<Map<String, Object>, String> keyed, CepRuleSpec spec) {
        return keyed
                .window(TumblingEventTimeWindows.of(parseWindowSize(spec)))
                .aggregate(new ThresholdAggFunction(spec));
    }

    /** 聚合函数：累加字段值，窗口结束时判断是否超阈值。 */
    static class ThresholdAggFunction
            implements AggregateFunction<Map<String, Object>, double[], CepFiredEvent> {

        private final CepRuleSpec spec;

        ThresholdAggFunction(CepRuleSpec spec) { this.spec = spec; }

        @Override public double[] createAccumulator() { return new double[]{0.0, 0}; }

        @Override
        public double[] add(Map<String, Object> event, double[] acc) {
            Object raw = event.get(spec.aggregateField);
            if (raw instanceof Number n) {
                acc[0] += n.doubleValue(); // SUM / 累加用于 AVG
                acc[1]++;
            }
            return acc;
        }

        @Override
        public CepFiredEvent getResult(double[] acc) {
            double value = switch (spec.aggregateFunc) {
                case "COUNT" -> acc[1];
                case "AVG"   -> acc[1] == 0 ? 0 : acc[0] / acc[1];
                default      -> acc[0]; // SUM / MAX / MIN（MAX/MIN 简化为 SUM，v2 再精化）
            };
            if (value <= spec.threshold) return null; // 未超阈值，不命中
            return new CepFiredEvent(
                    spec.tenantId, spec.sceneCode, "unknown_subject",
                    spec.ruleVersionId, spec.kind,
                    Map.of(spec.aggregateField, value), Instant.now());
        }

        @Override
        public double[] merge(double[] a, double[] b) {
            return new double[]{a[0] + b[0], a[1] + b[1]};
        }
    }
}
```

```java
// rule-cep-job/src/main/java/com/sstlfsj/rule/cep/pattern/SequencePatternFactory.java
package com.sstlfsj.rule.cep.pattern;

import com.sstlfsj.rule.cep.model.CepRuleSpec;
import org.apache.flink.cep.pattern.Pattern;
import org.apache.flink.cep.pattern.conditions.SimpleCondition;

import java.time.Duration;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.stream.Collectors;

/**
 * 序列类 CEP Pattern 构建器。
 * 按 steps 顺序串联 followedBy，not 事件用 notFollowedBy 插入 between 区间。
 */
public class SequencePatternFactory {

    /**
     * 构建序列类 Pattern。
     * steps 至少 2 个；notEvents 按 between[0]→between[1] 区间插入 notFollowedBy。
     *
     * @param spec CEP 规则描述（kind=CEP_SEQUENCE）
     * @return Flink Pattern
     */
    public static Pattern<Map<String, Object>, Map<String, Object>> build(CepRuleSpec spec) {
        Duration window = Duration.parse(spec.withinWindow);

        // 构建 notEvents 的别名区间索引，key = "afterAlias:beforeAlias"，value = notEventTypes
        Map<String, Set<String>> notMap = buildNotMap(spec.notEvents);

        List<CepRuleSpec.StepSpec> steps = spec.steps;
        // 第一个 step
        Pattern<Map<String, Object>, Map<String, Object>> pattern =
                Pattern.<Map<String, Object>>begin(steps.get(0).alias)
                        .where(eventTypeCondition(steps.get(0).eventType));

        // 后续 steps：在 steps[i-1] → steps[i] 之间插入 not 事件
        for (int i = 1; i < steps.size(); i++) {
            CepRuleSpec.StepSpec prev = steps.get(i - 1);
            CepRuleSpec.StepSpec curr = steps.get(i);

            // 插入 NOT 事件（between prev.alias 和 curr.alias）
            String notKey = prev.alias + ":" + curr.alias;
            Set<String> notTypes = notMap.get(notKey);
            if (notTypes != null) {
                for (String notType : notTypes) {
                    pattern = pattern.notFollowedBy("not_" + notType)
                            .where(eventTypeCondition(notType));
                }
            }
            pattern = pattern.followedBy(curr.alias)
                    .where(eventTypeCondition(curr.eventType));
        }

        return pattern.within(
                org.apache.flink.streaming.api.windowing.time.Time.milliseconds(window.toMillis()));
    }

    private static Map<String, Set<String>> buildNotMap(List<CepRuleSpec.NotEventSpec> notEvents) {
        if (notEvents == null) return Map.of();
        return notEvents.stream()
                .filter(n -> n.between != null && n.between.size() == 2)
                .collect(Collectors.groupingBy(
                        n -> n.between.get(0) + ":" + n.between.get(1),
                        Collectors.mapping(n -> n.eventType, Collectors.toSet())));
    }

    private static SimpleCondition<Map<String, Object>> eventTypeCondition(String eventType) {
        return new SimpleCondition<>() {
            @Override
            public boolean filter(Map<String, Object> event) {
                return eventType.equals(event.get("eventType"));
            }
        };
    }
}
```

```java
// rule-cep-job/src/main/java/com/sstlfsj/rule/cep/CepJobMain.java
package com.sstlfsj.rule.cep;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.sstlfsj.rule.cep.model.CepFiredEvent;
import com.sstlfsj.rule.cep.model.CepRuleSpec;
import com.sstlfsj.rule.cep.pattern.AggregationWindowFactory;
import com.sstlfsj.rule.cep.pattern.FrequencyPatternFactory;
import com.sstlfsj.rule.cep.pattern.SequencePatternFactory;
import com.sstlfsj.rule.cep.sink.CepFiredEventSink;
import com.sstlfsj.rule.cep.source.RocketMqEventSource;
import org.apache.flink.cep.CEP;
import org.apache.flink.cep.PatternSelectFunction;
import org.apache.flink.cep.PatternStream;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Instant;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

/**
 * Flink CEP Job 入口。
 * 环境变量：
 *   RULE_API_URL    — rule-api 地址，如 http://rule-api:8080
 *   ROCKETMQ_ADDR   — RocketMQ broker，如 localhost:9876
 *   TENANT_ID       — 租户 ID
 */
public class CepJobMain {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    public static void main(String[] args) throws Exception {
        String ruleApiUrl  = System.getenv().getOrDefault("RULE_API_URL", "http://localhost:8080");
        String rocketmqAddr = System.getenv().getOrDefault("ROCKETMQ_ADDR", "localhost:9876");
        String tenantId    = System.getenv().getOrDefault("TENANT_ID", "1");

        // 1. 启动时全量拉取 CEP 规则
        List<CepRuleSpec> rules = loadCepRules(ruleApiUrl, tenantId);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        // Watermark 策略：使用事件时间，允许 5 秒乱序
        env.getConfig().setAutoWatermarkInterval(1000);

        // 2. RocketMQ source（rule-events topic）
        DataStream<Map<String, Object>> eventStream = env
                .addSource(new RocketMqEventSource(rocketmqAddr, "rule-events"))
                .name("rocketmq-rule-events");

        List<DataStream<CepFiredEvent>> firedStreams = new ArrayList<>();

        // 3. 按规则 kind 分派到对应算子
        for (CepRuleSpec spec : rules) {
            DataStream<Map<String, Object>> filtered = eventStream
                    .filter(e -> spec.eventType.equals(e.get("eventType"))
                            && spec.tenantId.equals(e.get("tenantId"))
                            && spec.sceneCode.equals(e.get("sceneCode")))
                    .keyBy(e -> e.get("subjectId").toString());

            switch (spec.kind) {
                case "CEP_FREQUENCY" -> {
                    PatternStream<Map<String, Object>> ps = CEP.pattern(
                            filtered, FrequencyPatternFactory.build(spec));
                    firedStreams.add(ps.select(matchToFired(spec)));
                }
                case "CEP_SEQUENCE" -> {
                    PatternStream<Map<String, Object>> ps = CEP.pattern(
                            filtered, SequencePatternFactory.build(spec));
                    firedStreams.add(ps.select(matchToFired(spec)));
                }
                case "CEP_AGGREGATION" -> {
                    // 聚合类使用 window 算子，结果过滤 null（未超阈值的窗口）
                    DataStream<CepFiredEvent> agg = AggregationWindowFactory
                            .apply(filtered.keyBy(e -> e.get("subjectId").toString()), spec)
                            .filter(f -> f != null);
                    firedStreams.add(agg);
                }
            }
        }

        // 4. 合并所有命中流，写入 rule-cep-fired topic
        if (!firedStreams.isEmpty()) {
            DataStream<CepFiredEvent> merged = firedStreams.get(0);
            for (int i = 1; i < firedStreams.size(); i++) {
                merged = merged.union(firedStreams.get(i));
            }
            merged.addSink(new CepFiredEventSink(rocketmqAddr, "rule-cep-fired"))
                    .name("cep-fired-sink");
        }

        env.execute("rule-cep-job");
    }

    /** HTTP GET /api/v1/cep/rules，解析返回的 RuleVersionRow 列表为 CepRuleSpec。 */
    private static List<CepRuleSpec> loadCepRules(String apiUrl, String tenantId) throws Exception {
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest req = HttpRequest.newBuilder()
                .uri(URI.create(apiUrl + "/api/v1/cep/rules?tenantId=" + tenantId))
                .GET().build();
        HttpResponse<String> resp = client.send(req, HttpResponse.BodyHandlers.ofString());
        // 响应结构：{ "data": [ { "conditionAstJson": "...", "ruleVersionId": 1, "tenantId": "...", "sceneCode": "..." } ] }
        var root = MAPPER.readTree(resp.body());
        var data = root.get("data");
        List<CepRuleSpec> specs = new ArrayList<>();
        if (data != null && data.isArray()) {
            for (var row : data) {
                String specJson = row.get("conditionAstJson").asText();
                CepRuleSpec spec = MAPPER.readValue(specJson, CepRuleSpec.class);
                spec.ruleVersionId = row.get("ruleVersionId").asLong();
                spec.tenantId = row.get("tenantId").asText();
                spec.sceneCode = row.get("sceneCode").asText();
                specs.add(spec);
            }
        }
        return specs;
    }

    /** 将 Flink CEP 匹配结果转换为 CepFiredEvent。 */
    private static PatternSelectFunction<Map<String, Object>, CepFiredEvent> matchToFired(
            CepRuleSpec spec) {
        return match -> {
            // 取第一个 step 的事件 payload 作为 matchContext
            Map<String, Object> firstEvent = match.values().stream()
                    .flatMap(List::stream).findFirst().orElse(Map.of());
            String subjectId = firstEvent.getOrDefault("subjectId", "unknown").toString();
            return new CepFiredEvent(
                    spec.tenantId, spec.sceneCode, subjectId,
                    spec.ruleVersionId, spec.kind,
                    firstEvent, Instant.now());
        };
    }
}
```
